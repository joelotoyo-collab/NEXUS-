const express = require("express");
const OpenAI = require("openai");
const path = require("path");
require("dotenv").config();

const app = express();
const PORT = 3000;

const client = new OpenAI({
  apiKey: process.env.OPENROUTER_API_KEY,
  baseURL: "https://openrouter.ai/api/v1"
});

app.use(express.json());
app.use(express.static(__dirname));

app.post("/api/chat", async (req, res) => {
  try {
    const { message } = req.body;

    if (!message || !message.trim()) {
      return res.status(400).json({
        error: "Message cannot be empty."
      });
    }

    const response = await client.chat.completions.create({
      model: "openrouter/free",
      messages: [
        {
          role: "system",
          content:
            "You are NEXUS, a futuristic personal AI command assistant. " +
            "Call the user Joel or Commander. Be intelligent, helpful, concise, " +
            "and occasionally playful."
        },
        {
          role: "user",
          content: message
        }
      ]
    });

    res.json({
      reply: response.choices[0].message.content
    });

  } catch (error) {
    console.error("NEXUS AI ERROR:", error);

    res.status(500).json({
      error: "NEXUS could not reach the AI system."
    });
  }
});

app.use((req, res) => {
  res.sendFile(path.join(__dirname, "index.html"));
});

app.listen(PORT, "127.0.0.1", () => {
  console.log("");
  console.log("=================================");
  console.log("      NEXUS V1.2 ONLINE");
  console.log("=================================");
  console.log(`COMMAND CENTER: http://localhost:${PORT}`);
  console.log("AI BACKEND: READY");
  console.log("=================================");
});
